  Future<void> openAccountingSalesInvoicePage({
    Map<String, dynamic>? existingSale,cd ؤيؤي
    final isArabic = widget.currentUser.language == 'ar';

    try {
      final customersData = await Supabase.instance.client
          .from('accounting_customers')
          .select()
          .eq('created_by', widget.currentUser.id)
          .eq('active', true)
          .order('name');

      final salesData = await Supabase.instance.client
          .from('accounting_sales')
          .select('invoice_number')
          .eq('created_by', widget.currentUser.id)
          .order('invoice_number', ascending: false)
          .limit(1);

      final productsData = await Supabase.instance.client
          .from('accounting_products')
          .select()
          .eq('created_by', widget.currentUser.id)
          .eq('active', true)
          .order('name');

      if (!mounted) return;

      final customers = List<Map<String, dynamic>>.from(customersData);
      final products = List<Map<String, dynamic>>.from(productsData);

      int nextInvoiceNumber = 1;

      if (salesData.isNotEmpty) {
        final lastNumber =
            (salesData.first['invoice_number'] as num?)?.toInt() ?? 0;
        nextInvoiceNumber = lastNumber + 1;
      }

      final customerSearchController = TextEditingController();

      Map<String, dynamic>? selectedCustomer;
      List<Map<String, dynamic>> filteredCustomers = [];

      double previousBalanceUsd = 0;
      double previousBalanceSyp = 0;
      bool searchingCustomer = false;
      bool initialEditBalanceLoaded = false;

      final invoiceItems = <Map<String, dynamic>>[];
      bool savingInvoice = false;

      final invoiceDiscountUsdController = TextEditingController(text: '0');
      final invoiceDiscountSypController = TextEditingController(text: '0');
      final paidUsdController = TextEditingController(text: '0');
      final paidSypController = TextEditingController(text: '0');

      final isEditingSale = existingSale != null;

      if (isEditingSale) {
        final saleId = existingSale['id']?.toString();

        if (saleId == null || saleId.isEmpty) {
          throw Exception(
            isArabic ? 'معرّف الفاتورة غير موجود' : 'Invoice ID is missing',
          );
        }

        final customerId = existingSale['customer_id']?.toString();

        if (customerId != null && customerId.isNotEmpty) {
          for (final customer in customers) {
            if (customer['id']?.toString() == customerId) {
              selectedCustomer = customer;
              customerSearchController.text =
                  customer['name']?.toString() ?? '';
              searchingCustomer = true;
              break;
            }
          }
        }

        paidUsdController.text =
            ((existingSale['paid_usd'] as num?)?.toDouble() ?? 0)
                .toStringAsFixed(2);

        paidSypController.text =
            ((existingSale['paid_syp'] as num?)?.toDouble() ?? 0)
                .toStringAsFixed(0);

        final savedItemsData = await Supabase.instance.client
            .from('accounting_sale_items')
            .select()
            .eq('sale_id', saleId);

        final savedItems = List<Map<String, dynamic>>.from(savedItemsData);

        for (final savedItem in savedItems) {
          final productId = savedItem['product_id']?.toString();

          Map<String, dynamic>? product;

          for (final item in products) {
            if (item['id']?.toString() == productId) {
              product = item;
              break;
            }
          }

          if (product == null) {
            continue;
          }

          invoiceItems.add({
            'product': product,
            'searchController': TextEditingController(
              text: product['name']?.toString() ?? '',
            ),
            'quantityController': TextEditingController(
              text: ((savedItem['quantity'] as num?)?.toDouble() ?? 0)
                  .toString(),
            ),
            'priceController': TextEditingController(
              text: ((savedItem['unit_price'] as num?)?.toDouble() ?? 0)
                  .toString(),
            ),
            'discountController': TextEditingController(
              text: ((savedItem['discount'] as num?)?.toDouble() ?? 0)
                  .toString(),
            ),
            'priceCurrency': savedItem['currency']?.toString() ?? 'USD',
            'filteredProducts': <Map<String, dynamic>>[],
          });
        }
      }

      Map<String, dynamic> createEmptyInvoiceRow() {
        return {
          'product': null,
          'searchController': TextEditingController(),
          'quantityController': TextEditingController(text: '1'),
          'priceController': TextEditingController(),
          'discountController': TextEditingController(text: '0'),
          'priceCurrency': 'USD',
          'filteredProducts': <Map<String, dynamic>>[],
        };
      }

      double rowTotal(Map<String, dynamic> row) {
        final quantity =
            double.tryParse(row['quantityController'].text.trim()) ?? 0;

        final price = double.tryParse(row['priceController'].text.trim()) ?? 0;

        var discount =
            double.tryParse(row['discountController'].text.trim()) ?? 0;

        if (discount < 0) discount = 0;
        if (discount > 100) discount = 100;

        final beforeDiscount = quantity * price;

        return beforeDiscount * (1 - discount / 100);
      }

      double invoiceItemsTotal(String currency) {
        double total = 0;

        for (final row in invoiceItems) {
          if (row['product'] == null) continue;

          final rowCurrency = row['priceCurrency']?.toString() ?? 'USD';

          if (rowCurrency == currency) {
            total += rowTotal(row);
          }
        }

        return total;
      }

      double invoiceDiscountPercent(TextEditingController controller) {
        var value = double.tryParse(controller.text.trim()) ?? 0;

        if (value < 0) value = 0;
        if (value > 100) value = 100;

        return value;
      }

      double invoiceNetTotal(
        String currency,
        TextEditingController discountController,
      ) {
        final total = invoiceItemsTotal(currency);
        final discount = invoiceDiscountPercent(discountController);

        return total * (1 - discount / 100);
      }

              Future<void> selectCustomer(Map<String, dynamic> customer) async {
                setDialogState(() {
                  selectedCustomer = customer;
                  customerSearchController.text =
                      customer['name']?.toString() ?? '';
                  filteredCustomers = [];
                  searchingCustomer = true;
                });

                try {
                  final customerId = customer['id'].toString();

                  final transactions = await Supabase.instance.client
                      .from('accounting_transactions')
                      .select()
                      .eq('customer_id', customerId);

                  final sales = await Supabase.instance.client
                      .from('accounting_sales')
                      .select()
                      .eq('created_by', widget.currentUser.id)
                      .eq('customer_id', customerId);

                  final openingBalance =
                      (customer['opening_balance'] as num?)?.toDouble() ?? 0;

                  final customerCurrency =
                      customer['currency']?.toString() ?? 'USD';

                  double balanceUsd = customerCurrency == 'USD'
                      ? openingBalance
                      : 0;

                  double balanceSyp = customerCurrency == 'SYP'
                      ? openingBalance
                      : 0;

                  final editingSaleId = existingSale?['id']?.toString();

                  for (final sale in sales) {
                    if (sale['status']?.toString() == 'cancelled') {
                      continue;
                    }

                    // عند تعديل فاتورة لا نحسب نفس الفاتورة
                    // ضمن الرصيد السابق.
                    if (editingSaleId != null &&
                        sale['id']?.toString() == editingSaleId) {
                      continue;
                    }

                    final totalUsd =
                        (sale['total_usd'] as num?)?.toDouble() ?? 0;

                    final totalSyp =
                        (sale['total_syp'] as num?)?.toDouble() ?? 0;

                    final paidUsd = (sale['paid_usd'] as num?)?.toDouble() ?? 0;

                    final paidSyp = (sale['paid_syp'] as num?)?.toDouble() ?? 0;

                    balanceUsd += totalUsd - paidUsd;
                    balanceSyp += totalSyp - paidSyp;
                  }

                  // الحركات القديمة ما زالت مرتبطة بعملة واحدة.
                  // نطبقها على عملة العميل فقط حتى لا نخلط العملات.
                  for (final transaction in transactions) {
                    final type = transaction['transaction_type']?.toString();

                    final amount =
                        (transaction['amount'] as num?)?.toDouble() ?? 0;

                    double effect = 0;

                    if (type == 'customer_receipt') {
                      effect = -amount;
                    } else if (type == 'customer_payment') {
                      effect = amount;
                    } else {
                      continue;
                    }

                    if (customerCurrency == 'SYP') {
                      balanceSyp += effect;
                    } else {
                      balanceUsd += effect;
                    }
                  }

                  if (!dialogContext.mounted) return;

                  setDialogState(() {
                    previousBalanceUsd = balanceUsd;
                    previousBalanceSyp = balanceSyp;
                    searchingCustomer = false;
                  });
                } catch (e) {
                  if (!dialogContext.mounted) return;

                  setDialogState(() {
                    searchingCustomer = false;
                  });

                  ScaffoldMessenger.of(context).showSnackBar(
                    SnackBar(
                      content: Text(
                        isArabic ? 'تعذر تحديث رصيد العميل، حاول مرة أخرى' : 'Unable to update customer balance. Please try again.',
                      ),
                    ),
                  );
                }
              }

              if (isEditingSale &&
                  selectedCustomer != null &&
                  !initialEditBalanceLoaded) {
                initialEditBalanceLoaded = true;

                WidgetsBinding.instance.addPostFrameCallback((_) {
                  if (!dialogContext.mounted) return;
                  selectCustomer(selectedCustomer!);
                });
              }

              Future<void> saveAccountingSalesInvoice() async {
                if (savingInvoice) return;

                if (selectedCustomer == null) {
                  ScaffoldMessenger.of(context).showSnackBar(
                    SnackBar(
                      content: Text(
                        isArabic
                            ? 'اختر العميل أولاً'
                            : 'Select a customer first',
                      ),
                    ),
                  );
                  return;
                }

                final validItems = invoiceItems.where((row) {
                  return row['product'] != null;
                }).toList();

                if (validItems.isEmpty) {
                  ScaffoldMessenger.of(context).showSnackBar(
                    SnackBar(
                      content: Text(
                        isArabic
                            ? 'أضف مادة واحدة على الأقل إلى الفاتورة'
                            : 'Add at least one invoice item',
                      ),
                    ),
                  );
                  return;
                }

                for (final row in validItems) {
                  final product = row['product'] as Map<String, dynamic>?;

                  if (product == null) continue;

                  final quantity =
                      double.tryParse(row['quantityController'].text.trim()) ??
                      0;

                  final stock = (product['quantity'] as num?)?.toDouble() ?? 0;

                  if (quantity <= 0) {
                    ScaffoldMessenger.of(context).showSnackBar(
                      SnackBar(
                        content: Text(
                          isArabic
                              ? 'توجد مادة بكمية غير صحيحة'
                              : 'An item has an invalid quantity',
                        ),
                      ),
                    );
                    return;
                  }

                  if (quantity > stock) {
                    ScaffoldMessenger.of(context).showSnackBar(
                      SnackBar(
                        content: Text(
                          isArabic
                              ? 'الكمية المطلوبة للمادة ${product['name'] ?? ''} أكبر من المخزون'
                              : 'Requested quantity for ${product['name'] ?? ''} exceeds stock',
                        ),
                      ),
                    );
                    return;
                  }
                }

                setDialogState(() {
                  savingInvoice = true;
                });

                String? createdSaleId;

                try {
                  final usdTotal = invoiceItemsTotal('USD');
                  final sypTotal = invoiceItemsTotal('SYP');

                  final hasUsd = validItems.any(
                    (row) =>
                        (row['priceCurrency']?.toString() ?? 'USD') == 'USD',
                  );

                  final hasSyp = validItems.any(
                    (row) =>
                        (row['priceCurrency']?.toString() ?? 'USD') == 'SYP',
                  );

                  final legacyCurrency = hasSyp && !hasUsd ? 'SYP' : 'USD';

                  final legacyTotal = legacyCurrency == 'SYP'
                      ? sypTotal
                      : usdTotal;

                  final paidUsd =
                      (double.tryParse(paidUsdController.text.trim()) ?? 0)
                          .clamp(0, double.infinity);

                  final paidSyp =

      final suppliersData = await Supabase.instance.client
          .from('accounting_suppliers')

          .eq('active', true)
          .order('name');

      final productsData = await Supabase.instance.client
          .from('accounting_products')
          .select()
          .eq('created_by', widget.currentUser.id)
          .eq('active', true)
          .order('name');

      final purchasesData = await Supabase.instance.client
          .from('accounting_purchases')
          .select('invoice_number')
          .eq('created_by', widget.currentUser.id)
          .order('invoice_number', ascending: false)
          .limit(1);

      if (!mounted) return;

      final suppliers = List<Map<String, dynamic>>.from(suppliersData);
      final products = List<Map<String, dynamic>>.from(productsData);

      if (suppliers.isEmpty) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(
              isArabic
                  ? 'يجب إضافة مورد فعال أولًا'
                  : 'Please add an active supplier first',
            ),
          ),
        );
        return;
      }

      if (products.isEmpty) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(
              isArabic
                  ? 'يجب إضافة مادة فعالة أولًا'
                  : 'Please add an active product first',
            ),
          ),
        );
        return;
      }

      int nextInvoiceNumber = 1;

      if (purchasesData.isNotEmpty) {
        final lastNumber =
            (purchasesData.first['invoice_number'] as num?)?.toInt() ?? 0;
        nextInvoiceNumber = lastNumber + 1;
      }

      bool savingInvoice = false;

      double parseNumber(String value) {
        var text = value.trim().replaceAll(',', '');

        if (text.isEmpty) return 0;

        final direct = double.tryParse(text);

        if (direct != null) return direct;

        if (text.contains('*')) {
          final parts = text.split('*');

          double result = 1;

          for (final part in parts) {
            final number = double.tryParse(part.trim());

            if (number == null) return 0;

            result *= number;
          }

          return result;
        }

        return 0;
      }

        final quantity = parseNumber(
          (item['quantityController'] as TextEditingController).text,
        );

        final price = parseNumber(
          (item['priceController'] as TextEditingController).text,
        );

        final discount = parseNumber(
          (item['discountController'] as TextEditingController).text,
        );

          return beforeDiscount;
        }

        return beforeDiscount * (1 - (discount / 100));

                } else {
                  totalUsd += total;
                }
              }

              final remainingSyp = totalSyp - paidSyp;

                              final purchaseData = await Supabase

                                  'discount': discountPercent,

                              for (final item in invoiceItems) {
                                final productId = item['product_id']

                                }

                                final quantity = parseNumber(
                                  (item['quantityController']
                                          as TextEditingController)
                                      .text,

                                if (quantity <= 0) {
                                  continue;
                                }

                                final freshProduct = await Supabase
                                    .instance
                                    .client
                                    .from('accounting_products')
                                    .select('quantity')
                                    .eq('id', productId)
                                    .eq('created_by', widget.currentUser.id)
                                    .single();

                                final quantityBefore =
                                    (freshProduct['quantity'] as num?)
                                        ?.toDouble() ??
                                    0;

                                final quantityAfter = quantityBefore + quantity;

                                await Supabase.instance.client
                                    .from('accounting_products')
                                    .update({
                                      'quantity': quantityAfter,
                                      'updated_at': DateTime.now()
                                          .toIso8601String(),
                                    })
                                    .eq('id', productId)
                                    .eq('created_by', widget.currentUser.id);

                                      .from('accounting_inventory_movements')

                                        'product_id': productId,
                                        'movement_type': 'purchase',
                                        'quantity': quantity,
                                        'quantity_before': quantityBefore,
                                        'quantity_after': quantityAfter,
                                        'reference_id': purchaseId,
                                        'notes': isArabic
                                            ? 'شراء - فاتورة رقم $nextInvoiceNumber'

                                  // نعيد كمية المادة كما كانت.
                                  await Supabase.instance.client
                                      .from('accounting_products')
                                      .update({
                                        'quantity': quantityBefore,
                                        'updated_at': DateTime.now()
                                            .toIso8601String(),
                                      })
                                      .eq('id', productId)
                                      .eq('created_by', widget.currentUser.id);

                                  rethrow;
                                }
                              }

                              await loadAccountingProducts();
                              await loadAccountingPurchases();

                              if (!mounted) return;

                              Navigator.pop(dialogContext);

                              ScaffoldMessenger.of(context).showSnackBar(
                                SnackBar(
                                  content: Text(
                                    isArabic
                                        ? 'تم حفظ فاتورة المشتريات بنجاح'
                                        : 'Purchase invoice saved successfully',
                                  ),
                                ),
                              );
                            } catch (e) {
                              debugPrint('SAVE ACCOUNTING PURCHASE ERROR: $e');

                              if (!mounted) return;

                              setDialogState(() {
                                savingInvoice = false;
                              });

                              ScaffoldMessenger.of(context).showSnackBar(
                                SnackBar(
                                  content: Text(
                                    isArabic
                                        ? 'تعذر حفظ فاتورة المشتريات: $e'
                                        : 'Unable to save purchase invoice: $e',
                                  ),
                                  backgroundColor: Colors.red,
                                ),
                              );
                            }
                          },
                    icon: savingInvoice
                        ? const SizedBox(
                            width: 18,
                            height: 18,
                            child: CircularProgressIndicator(strokeWidth: 2),
                          )
                        : const Icon(Icons.save),
                    label: Text(isArabic ? 'حفظ الفاتورة' : 'Save Invoice'),
                  ),
                ],
              );
            },
          );
        },
      );
    } catch (e) {
      debugPrint('OPEN ACCOUNTING PURCHASE ERROR: $e');

      if (!mounted) return;

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            isArabic
                ? 'تعذر فتح فاتورة المشتريات: $e'
                : 'Unable to open purchase invoice: $e',
          ),
          backgroundColor: Colors.red,
        ),
      );
    }
